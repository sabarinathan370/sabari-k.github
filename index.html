ID: " << ticketID << endl; } void saveTicket() { ofstream file("ticket.dat", ios::app); file.write((char*)this, sizeof(*this)); file.close(); } static void displayTickets() { Ticket t; ifstream file("ticket.dat"); while (file.read((char*)&t, sizeof(t))) { cout << "\nTicket ID: " << t.ticketID << "\nBus No: " << t.busNo << "\nPassenger: " << t.passenger << "\nSeat No: " << t.seatNo << endl; } file.close(); } static void cancelTicket(int id) { Ticket t; ifstream file("ticket.dat"); ofstream temp("temp.dat"); while (file.read((char*)&t, sizeof(t))) { if (t.ticketID != id) { temp.write((char*)&t, sizeof(t)); } } file.close(); temp.close(); remove("ticket.dat"); rename("temp.dat", "ticket.dat"); cout << "Ticket Cancelled Successfully!\n"; } }; void showMenu() { cout << "\n===== BUS RESERVATION SYSTEM =====\n"; cout << "1. Add Bus\n"; cout << "2. View Buses\n"; cout << "3. Book Ticket\n"; cout << "4. View Tickets\n"; cout << "5. Cancel Ticket\n"; cout << "6. Exit\n"; cout << "Enter choice: "; } int main() { int choice; Bus b; Ticket t; do { showMenu(); cin >> choice; switch (choice) { case 1: b.installBus(); break; case 2: { ifstream file("bus.dat"); while (file.read((char*)&b, sizeof(b))) { b.displayBus(); } file.close(); break; } case 3: t.bookTicket(); break; case 4: Ticket::displayTickets(); break; case 5: { int id; cout << "Enter Ticket ID to cancel: "; cin >> id; Ticket::cancelTicket(id); break; } case 6: cout << "Exiting...\n"; break; default: cout << "Invalid choice!\n"; } } while (choice != 6); return 0; }
